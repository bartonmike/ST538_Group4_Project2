\begin{frame}{What we learned}
        \textbf{\Large Takeaways:}
        \begin{itemize}
        \item Income: Education, age, gender, and labor supply are key predictors
        \item Insurance: Large, persistent coverage gap for non-citizens
        \item Transportation: Useful but limited predictor of income
        \end{itemize}

        \textbf{
        Policy implication: Expanding coverage eligibility could reduce disparities \\
        Data source: U.S. Census Bureau, ACS 2020–2024
